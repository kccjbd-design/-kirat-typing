function submitTest() { clearInterval(timer); let l = document.getElementById("lang").value; let c = document.getElementById("cat").value; let s = document.getElementById("lessonSelect"); let currentText = data[l][c][s.value].text; let origWords = currentText.trim().split(/\s+/); let typedWords = document.getElementById("input").value.trim().split(/\s+/); let resDisplayHtml = ""; let oIdx = 0; // Original index let tIdx = 0; // Typed index let stats = { correct: 0, wrong: 0, missing: 0, extra: 0 }; // Alignment Algorithm while (oIdx < origWords.length || tIdx < typedWords.length) { let o = origWords[oIdx]; let t = typedWords[tIdx]; if (o === t) { resDisplayHtml += `${o} `; stats.correct++; oIdx++; tIdx++; } else if (t === undefined) { // Word Missing resDisplayHtml += `${o} `; stats.missing++; oIdx++; } else if (o === undefined) { // Extra word resDisplayHtml += `${t} `; stats.extra++; tIdx++; } else { // Try to re-sync if (t === origWords[oIdx + 1]) { // User skipped a word resDisplayHtml += `${o} `; stats.missing++; oIdx++; } else if (typedWords[tIdx + 1] === o) { // User added extra word resDisplayHtml += `${t} `; stats.extra++; tIdx++; } else { // Simple mismatch (Typo) resDisplayHtml += `${t} `; stats.wrong++; oIdx++; tIdx++; } } } // Accuracy Calculation let total = origWords.length; let acc = Math.round(((stats.correct) / total) * 100); if(acc < 0) acc = 0; // Display Stats let statsHtml = `

Report Summary

Accuracy: ${acc}%

✅ Correct: ${stats.correct}

❌ Wrong/Typos: ${stats.wrong}

⚠️ Missing: ${stats.missing}

➕ Extra Words: ${stats.extra}

`; document.getElementById("resAccuracy").innerHTML = statsHtml; document.getElementById("resDisplay").innerHTML = resDisplayHtml; document.getElementById("modal").style.display = "block"; }
