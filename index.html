function submitTest() { clearInterval(timer); let l = document.getElementById("lang").value; let c = document.getElementById("cat").value; let s = document.getElementById("lessonSelect"); // Safety check if data exists if (!data[l] || !data[l][c] || !data[l][c][s.value]) return; let currentText = data[l][c][s.value].text; let origWords = currentText.trim().split(/\s+/); let typedWords = document.getElementById("input").value.trim().split(/\s+/); let resDisplayHtml = ""; let stats = { correct: 0, wrong: 0, missing: 0, extra: 0 }; // Linear Comparison: Compare word by word without "re-syncing" // This prevents the cascading error effect seen in screenshots let maxLength = Math.max(origWords.length, typedWords.length); for (let i = 0; i < maxLength; i++) { let o = origWords[i]; let t = typedWords[i]; if (t === undefined) { // Missing word resDisplayHtml += `${o} `; stats.missing++; } else if (o === undefined) { // Extra word resDisplayHtml += `${t} `; stats.extra++; } else if (o === t) { // Correct resDisplayHtml += `${t} `; stats.correct++; } else { // Wrong word (Mismatch) resDisplayHtml += `${t} `; stats.wrong++; } } // Accuracy Calculation let acc = Math.round((stats.correct / origWords.length) * 100); if(isNaN(acc)) acc = 0; // Report Summary let statsHtml = `

📊 ਰਿਜ਼ਲਟ ਰਿਪੋਰਟ

Accuracy: ${acc}%

✅ ਸਹੀ (Correct): ${stats.correct}

❌ ਗਲਤ (Wrong): ${stats.wrong}

⚠️ ਛੱਡੇ ਗਏ (Missing): ${stats.missing}

➕ ਵਾਧੂ (Extra): ${stats.extra}

`; document.getElementById("resAccuracy").innerHTML = statsHtml; document.getElementById("resDisplay").innerHTML = resDisplayHtml; document.getElementById("modal").style.display = "block"; }
