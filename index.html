// --- SMART ALIGNMENT LOGIC --- function submitTest() { clearInterval(timer); let l = document.getElementById("lang").value; let c = document.getElementById("cat").value; let s = document.getElementById("lessonSelect"); let currentText = data[l][c][s.value].text; let origWords = currentText.trim().split(/\s+/); let typedWords = document.getElementById("input").value.trim().split(/\s+/); let resDisplayHtml = ""; let oIdx = 0; // Original Word Index let tIdx = 0; // Typed Word Index let correctCount = 0; while (oIdx < origWords.length || tIdx < typedWords.length) { let o = origWords[oIdx]; let t = typedWords[tIdx]; if (o !== undefined && t !== undefined && o === t) { // ਸਹੀ ਸ਼ਬਦ (Perfect Match) resDisplayHtml += `${o} `; correctCount++; oIdx++; tIdx++; } else { // Mismatch: Check for Extra or Missing to Re-sync if (o === undefined) { // ਵਾਧੂ ਸ਼ਬਦ (Extra word typed) resDisplayHtml += `${t} `; tIdx++; } else if (t === undefined) { // ਸ਼ਬਦ ਮਿਸ ਹੋ ਗਿਆ (Missing word) resDisplayHtml += `${o} `; oIdx++; } else { // ਕੋਈ ਗਲਤੀ ਹੈ, ਦੇਖੋ ਕਿ ਕੀ ਅਗਲੇ ਸ਼ਬਦ ਨਾਲ Sync ਹੋ ਸਕਦਾ ਹੈ if (t === origWords[oIdx + 1]) { // ਯੂਜ਼ਰ ਨੇ ਵਰਤਮਾਨ ਸ਼ਬਦ ਛੱਡ ਦਿੱਤਾ (Missing) resDisplayHtml += `${o} `; oIdx++; } else if (typedWords[tIdx + 1] === o) { // ਯੂਜ਼ਰ ਨੇ ਫਾਲਤੂ ਸ਼ਬਦ ਲਿਖਿਆ (Extra) resDisplayHtml += `${t} `; tIdx++; } else { // ਸਧਾਰਨ ਗਲਤ ਸ਼ਬਦ (Wrong word) resDisplayHtml += `${t} (${o}) `; oIdx++; tIdx++; } } } } let acc = Math.round((correctCount / origWords.length) * 100); document.getElementById("resAccuracy").innerText = "Accuracy: " + (isNaN(acc) ? 0 : acc) + "%"; document.getElementById("resDisplay").innerHTML = resDisplayHtml; document.getElementById("modal").style.display = "block"; }
