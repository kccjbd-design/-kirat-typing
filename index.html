function submitTest() { clearInterval(timer); let l = document.getElementById("lang").value; let c = document.getElementById("cat").value; let s = document.getElementById("lessonSelect"); let currentText = data[l][c][s.value].text; let origWords = currentText.trim().split(/\s+/); let typedWords = document.getElementById("input").value.trim().split(/\s+/); let resDisplayHtml = ""; let stats = { correct: 0, wrong: 0, missing: 0, extra: 0 }; let oIdx = 0; let tIdx = 0; // Alignment Loop while (oIdx < origWords.length || tIdx < typedWords.length) { let o = origWords[oIdx]; let t = typedWords[tIdx]; // 1. ਜੇ ਸ਼ਬਦ ਬਿਲਕੁਲ ਸਹੀ ਹੈ if (o !== undefined && t !== undefined && o === t) { resDisplayHtml += `${t} `; stats.correct++; oIdx++; tIdx++; } // 2. ਜੇ ਟਾਈਪ ਕੀਤਾ ਸ਼ਬਦ ਖਤਮ ਹੋ ਗਿਆ ਪਰ ਅਸਲ ਵਿੱਚ ਸ਼ਬਦ ਬਚੇ ਹਨ (Missing words) else if (t === undefined) { resDisplayHtml += `${o} `; stats.missing++; oIdx++; } // 3. ਜੇ ਅਸਲ ਟੈਕਸਟ ਖਤਮ ਹੋ ਗਿਆ ਪਰ ਤੁਸੀਂ ਵਾਧੂ ਟਾਈਪ ਕੀਤਾ ਹੈ (Extra words) else if (o === undefined) { resDisplayHtml += `${t} `; stats.extra++; tIdx++; } // 4. ਮੇਲ ਨਹੀਂ ਖਾਂਦਾ (Mismatch: ਹੋ ਸਕਦਾ ਹੈ ਗਲਤ ਟਾਈਪ ਕੀਤਾ ਹੋਵੇ ਜਾਂ ਕੋਈ ਸ਼ਬਦ ਛੱਡ ਦਿੱਤਾ ਹੋਵੇ) else { // Check: ਕੀ ਇਹ ਸ਼ਬਦ ਸਿਰਫ ਗਲਤ (Typo) ਹੈ? // ਅਸੀਂ ਦੇਖਦੇ ਹਾਂ ਕਿ ਕੀ ਅਗਲਾ ਸ਼ਬਦ ਮੈਚ ਕਰਦਾ ਹੈ if (typedWords[tIdx + 1] === origWords[oIdx + 1]) { // ਇਹ ਸਿਰਫ ਇੱਕ ਗਲਤ ਸ਼ਬਦ (Typo) ਹੈ resDisplayHtml += `${t} `; stats.wrong++; oIdx++; tIdx++; } // Check: ਕੀ ਯੂਜ਼ਰ ਨੇ ਕੋਈ ਸ਼ਬਦ ਛੱਡ ਦਿੱਤਾ ਹੈ (Missing)? else if (t === origWords[oIdx + 1]) { resDisplayHtml += `${o} `; stats.missing++; oIdx++; } // Check: ਕੀ ਯੂਜ਼ਰ ਨੇ ਕੋਈ ਸ਼ਬਦ ਵਾਧੂ ਲਿਖ ਦਿੱਤਾ ਹੈ (Extra)? else if (typedWords[tIdx + 1] === o) { resDisplayHtml += `${t} `; stats.extra++; tIdx++; } // ਜੇ ਕੁਝ ਵੀ ਸਮਝ ਨਾ ਆਵੇ ਤਾਂ ਸਿਰਫ ਗਲਤ ਮਾਰਕ ਕਰੋ else { resDisplayHtml += `${t} `; stats.wrong++; oIdx++; tIdx++; } } } // Accuracy calculation let totalWords = origWords.length; let acc = Math.round((stats.correct / totalWords) * 100); if (isNaN(acc)) acc = 0; // Report Summary Display let summary = `

Result Report

Accuracy: ${acc}%

✅ ਸਹੀ ਸ਼ਬਦ (Correct): ${stats.correct}

❌ ਗਲਤ ਸ਼ਬਦ (Wrong/Typo): ${stats.wrong}

⚠️ ਛੱਡੇ ਗਏ ਸ਼ਬਦ (Missing): ${stats.missing}

➕ ਵਾਧੂ ਸ਼ਬਦ (Extra): ${stats.extra}

`; document.getElementById("resAccuracy").innerHTML = summary; document.getElementById("resDisplay").innerHTML = resDisplayHtml; document.getElementById("modal").style.display = "block"; }
